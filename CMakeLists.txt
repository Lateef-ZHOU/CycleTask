cmake_minimum_required(VERSION 3.28)
project(LidarCycleTask LANGUAGES CXX)
find_package(Qt6 6.11.2 REQUIRED COMPONENTS Quick Qml GuiPrivate ShaderTools)
set(CMAKE_AUTOMOC ON)
include(cmake/Dependencies.cmake)
file(GLOB_RECURSE headers CONFIGURE_DEPENDS "include/*.h")
file(GLOB_RECURSE views CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "include/*.qml")
foreach(view IN LISTS views)
    get_filename_component(name "${view}" NAME)
    set_source_files_properties("${view}" PROPERTIES QT_RESOURCE_ALIAS "${name}")
endforeach()
set(import_paths)
foreach(module Lidar::UICore Lidar::ChartCore Lidar::TaskUI)
    get_target_property(output ${module} QT_QML_MODULE_OUTPUT_DIRECTORY)
    get_filename_component(import_root "${output}/../.." ABSOLUTE)
    list(APPEND import_paths "${import_root}")
endforeach()
qt_add_qml_module(LidarCycleTask STATIC URI Lidar.CycleTask VERSION 1.0
    CLASS_NAME LidarCycleTaskPlugin RESOURCE_PREFIX /qt/qml
    OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/Lidar/CycleTask"
    DEPENDENCIES Lidar.UI Lidar.Charts Lidar.TaskUI
    IMPORT_PATH ${import_paths}
    SOURCES ${headers} QML_FILES ${views})
add_library(Lidar::CycleTask ALIAS LidarCycleTask)
target_include_directories(LidarCycleTask PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include"
    PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include/Task/Cycle")
target_link_libraries(LidarCycleTask PUBLIC Lidar::TaskUI)
target_compile_features(LidarCycleTask PUBLIC cxx_std_20)
target_compile_definitions(LidarCycleTask PUBLIC QT_NO_KEYWORDS)
if(WIN32)
    target_compile_definitions(LidarCycleTask PUBLIC NOMINMAX)
endif()
if(MSVC)
    target_compile_options(LidarCycleTask PRIVATE /W4)
    get_target_property(sources LidarCycleTask SOURCES)
    foreach(source IN LISTS sources)
        if(source MATCHES "/[.]rcc/qmlcache/")
            set_source_files_properties("${source}" PROPERTIES COMPILE_OPTIONS /wd4702)
        endif()
    endforeach()
endif()
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
    if(BUILD_TESTING)
        add_subdirectory(tests)
    endif()
endif()
